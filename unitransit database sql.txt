-- 1. Create the database
CREATE DATABASE IF NOT EXISTS unitransit_db;
USE unitransit_db;

-- 2. Create independent tables (No Foreign Keys)

CREATE TABLE IF NOT EXISTS `admins` (
    `id` INT AUTO_INCREMENT,
    `username` VARCHAR(100) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `email` VARCHAR(150) NOT NULL,
    `clearance_level` INT NOT NULL DEFAULT 1,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_admin_username` (`username`),
    UNIQUE KEY `idx_admin_email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `passengers` (
    `id` INT AUTO_INCREMENT,
    `username` VARCHAR(100) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `phone` VARCHAR(20) DEFAULT NULL,
    `student_card_path` VARCHAR(255) DEFAULT NULL,
    `account_status` VARCHAR(50) DEFAULT 'pending',
    `bio` TEXT DEFAULT NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_passenger_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `drivers` (
    `id` INT AUTO_INCREMENT,
    `username` VARCHAR(100) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `phone` VARCHAR(20) DEFAULT NULL,
    `license_card_path` VARCHAR(255) DEFAULT NULL,
    `verification_status` VARCHAR(50) DEFAULT 'pending',
    `bio` TEXT DEFAULT NULL,
    `is_available` TINYINT(1) DEFAULT 1,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_driver_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `routes` (
    `id` INT AUTO_INCREMENT,
    `pickup_location` VARCHAR(255) NOT NULL,
    `dropoff_location` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- 3. Create dependent tables (With Foreign Keys)

CREATE TABLE IF NOT EXISTS `vehicles` (
    `id` INT AUTO_INCREMENT,
    `driver_id` INT NOT NULL,
    `vehicle_type` VARCHAR(50) NOT NULL,
    `model_info` VARCHAR(150) DEFAULT NULL,
    `plate_number` VARCHAR(30) NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_vehicle_plate` (`plate_number`),
    KEY `idx_vehicle_driver` (`driver_id`),
    CONSTRAINT `vehicles_ibfk_1` FOREIGN KEY (`driver_id`) REFERENCES `drivers` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `bookings` (
    `id` INT AUTO_INCREMENT,
    `passenger_id` INT NOT NULL,
    `driver_id` INT DEFAULT NULL,
    `route_id` INT NOT NULL,
    `quoted_fare` DECIMAL(10,2) NOT NULL,
    `status` VARCHAR(50) DEFAULT 'pending',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `vehicle_type` VARCHAR(50) DEFAULT NULL,
    PRIMARY KEY (`id`),
    KEY `idx_booking_route` (`route_id`),
    KEY `idx_booking_passenger` (`passenger_id`),
    KEY `idx_booking_driver` (`driver_id`),
    CONSTRAINT `bookings_ibfk_1` FOREIGN KEY (`passenger_id`) REFERENCES `passengers` (`id`) ON DELETE CASCADE,
    CONSTRAINT `bookings_ibfk_2` FOREIGN KEY (`driver_id`) REFERENCES `drivers` (`id`) ON DELETE SET NULL,
    CONSTRAINT `bookings_ibfk_3` FOREIGN KEY (`route_id`) REFERENCES `routes` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `complaints` (
    `id` INT AUTO_INCREMENT,
    `passenger_id` INT NOT NULL,
    `admin_id` INT DEFAULT NULL,
    `complaint_text` TEXT NOT NULL,
    `resolution_notes` TEXT DEFAULT NULL,
    `status` VARCHAR(50) DEFAULT 'open',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    KEY `idx_complaint_admin` (`admin_id`),
    KEY `idx_complaint_passenger` (`passenger_id`),
    CONSTRAINT `complaints_ibfk_1` FOREIGN KEY (`passenger_id`) REFERENCES `passengers` (`id`) ON DELETE CASCADE,
    CONSTRAINT `complaints_ibfk_2` FOREIGN KEY (`admin_id`) REFERENCES `admins` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `ride_history` (
    `id` INT AUTO_INCREMENT,
    `booking_id` INT NOT NULL,
    `passenger_name` VARCHAR(100) NOT NULL,
    `driver_name` VARCHAR(100) NOT NULL,
    `pickup_loc` VARCHAR(255) NOT NULL,
    `dropoff_loc` VARCHAR(255) NOT NULL,
    `vehicle_type` VARCHAR(50) NOT NULL,
    `final_fare` DECIMAL(10,2) NOT NULL,
    `settled_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    KEY `idx_history_booking` (`booking_id`),
    CONSTRAINT `ride_history_ibfk_1` FOREIGN KEY (`booking_id`) REFERENCES `bookings` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
